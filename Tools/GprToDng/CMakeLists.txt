cmake_minimum_required(VERSION 3.23)
project(GprToDng LANGUAGES CXX)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(ENGINE_BUILD "${REPO_ROOT}/out/build/vs2026-x64" CACHE PATH "Existing engine build containing the GoPro SDK libraries")
set(GPR_SOURCE "${REPO_ROOT}/Extern/3rdParty/gpr/gpr/source/lib")
add_executable(GprToDng main.cpp)
target_compile_features(GprToDng PRIVATE cxx_std_17)
target_compile_definitions(GprToDng PRIVATE GPR_READING=1 GPR_WRITING=1)
target_include_directories(GprToDng PRIVATE "${GPR_SOURCE}/gpr_sdk/public" "${GPR_SOURCE}/common/public")
foreach(lib gpr_sdk dng_sdk xmp_core vc5_decoder vc5_encoder vc5_common common md5_lib expat_lib)
  target_link_libraries(GprToDng PRIVATE "${ENGINE_BUILD}/Extern/3rdParty/gpr/gpr/source/lib/${lib}/$<CONFIG>/${lib}.lib")
endforeach()
install(TARGETS GprToDng RUNTIME DESTINATION bin)
