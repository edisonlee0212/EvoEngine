function(evoengine_discover_plugins out_plugins)
	file(GLOB plugin_cmake_files
		LIST_DIRECTORIES false
		CONFIGURE_DEPENDS
		"${CMAKE_CURRENT_SOURCE_DIR}/*/CMakeLists.txt")

	set(discovered_plugins "")
	foreach(plugin_cmake_file ${plugin_cmake_files})
		get_filename_component(plugin_dir "${plugin_cmake_file}" DIRECTORY)
		get_filename_component(plugin_name "${plugin_dir}" NAME)
		if (NOT plugin_name MATCHES "^\\.")
			list(APPEND discovered_plugins ${plugin_name})
		endif()
	endforeach()
	list(SORT discovered_plugins)
	set(${out_plugins} ${discovered_plugins} PARENT_SCOPE)
endfunction()

function(evoengine_load_plugin_metadata plugin_name out_default_enable out_supported_platforms out_dependencies)
	set(default_enable ON)
	set(supported_platforms ALL)
	set(dependencies "")
	set(plugin_metadata_file "${CMAKE_CURRENT_SOURCE_DIR}/${plugin_name}/PluginInfo.cmake")

	if (EXISTS "${plugin_metadata_file}")
		set(EVOENGINE_PLUGIN_DEFAULT_ENABLE ${default_enable})
		set(EVOENGINE_PLUGIN_SUPPORTED_PLATFORMS ${supported_platforms})
		set(EVOENGINE_PLUGIN_DEPENDS ${dependencies})
		include("${plugin_metadata_file}")
		set(default_enable ${EVOENGINE_PLUGIN_DEFAULT_ENABLE})
		set(supported_platforms ${EVOENGINE_PLUGIN_SUPPORTED_PLATFORMS})
		set(dependencies ${EVOENGINE_PLUGIN_DEPENDS})
	endif()

	set(${out_default_enable} ${default_enable} PARENT_SCOPE)
	set(${out_supported_platforms} ${supported_platforms} PARENT_SCOPE)
	set(${out_dependencies} ${dependencies} PARENT_SCOPE)
endfunction()

function(evoengine_plugin_platform_supported supported_platforms out_supported)
	if (NOT supported_platforms OR supported_platforms STREQUAL "ALL")
		set(${out_supported} ON PARENT_SCOPE)
		return()
	endif()

	set(is_supported OFF)
	foreach(platform_name ${supported_platforms})
		string(TOUPPER "${platform_name}" platform_name_upper)
		if ((platform_name_upper STREQUAL "WINDOWS" OR platform_name_upper STREQUAL "WIN32") AND WIN32)
			set(is_supported ON)
		elseif ((platform_name_upper STREQUAL "LINUX" OR platform_name_upper STREQUAL "UNIX") AND UNIX AND NOT APPLE)
			set(is_supported ON)
		elseif ((platform_name_upper STREQUAL "MACOS" OR platform_name_upper STREQUAL "APPLE") AND APPLE)
			set(is_supported ON)
		endif()
	endforeach()

	set(${out_supported} ${is_supported} PARENT_SCOPE)
endfunction()

function(register_evoengine_plugin plugin_name)
	get_property(registered_plugins GLOBAL PROPERTY EVOENGINE_REGISTERED_PLUGIN_NAMES)
	if (plugin_name IN_LIST registered_plugins)
		return()
	endif()

	get_property(registering_plugins GLOBAL PROPERTY EVOENGINE_REGISTERING_PLUGIN_NAMES)
	if (plugin_name IN_LIST registering_plugins)
		message(FATAL_ERROR "Plugin dependency cycle detected while registering [${plugin_name}].")
	endif()
	set_property(GLOBAL APPEND PROPERTY EVOENGINE_REGISTERING_PLUGIN_NAMES ${plugin_name})

	evoengine_load_plugin_metadata(${plugin_name} default_enable supported_platforms plugin_dependencies)
	set(option_name EVOENGINE_ENABLE_${plugin_name}_PLUGIN)
	option(${option_name} "Enable ${plugin_name} Plugin." ${default_enable})

	evoengine_plugin_platform_supported("${supported_platforms}" plugin_platform_supported)
	if (NOT plugin_platform_supported)
		message("Skipping [${plugin_name}] Plugin: supported platforms are [${supported_platforms}].")
		set_property(GLOBAL APPEND PROPERTY EVOENGINE_REGISTERED_PLUGIN_NAMES ${plugin_name})
		set(EvoEngine_Plugins ${EvoEngine_Plugins} PARENT_SCOPE)
		set(EVOENGINE_PLUGINS_INCLUDES ${EVOENGINE_PLUGINS_INCLUDES} PARENT_SCOPE)
		set(EVOENGINE_PLUGINS_PCHS ${EVOENGINE_PLUGINS_PCHS} PARENT_SCOPE)
		set(EVOENGINE_PLUGINS_DEFS ${EVOENGINE_PLUGINS_DEFS} PARENT_SCOPE)
		set(EVOENGINE_PLUGINS_RUNTIME_LIBRARIES ${EVOENGINE_PLUGINS_RUNTIME_LIBRARIES} PARENT_SCOPE)
		set(EVOENGINE_PLUGINS_PDBS ${EVOENGINE_PLUGINS_PDBS} PARENT_SCOPE)
		set(EVOENGINE_PLUGINS_RESOURCES ${EVOENGINE_PLUGINS_RESOURCES} PARENT_SCOPE)
		return()
	endif()

	if (${option_name})
		foreach(dependency_name ${plugin_dependencies})
			if (dependency_name IN_LIST EVOENGINE_DISCOVERED_PLUGINS)
				register_evoengine_plugin(${dependency_name})
			endif()
			if (NOT TARGET ${dependency_name}Plugin)
				message("[${plugin_name}] Plugin not loaded: dependency [${dependency_name}] Plugin is not available.")
				set_property(GLOBAL APPEND PROPERTY EVOENGINE_REGISTERED_PLUGIN_NAMES ${plugin_name})
				set(EvoEngine_Plugins ${EvoEngine_Plugins} PARENT_SCOPE)
				set(EVOENGINE_PLUGINS_INCLUDES ${EVOENGINE_PLUGINS_INCLUDES} PARENT_SCOPE)
				set(EVOENGINE_PLUGINS_PCHS ${EVOENGINE_PLUGINS_PCHS} PARENT_SCOPE)
				set(EVOENGINE_PLUGINS_DEFS ${EVOENGINE_PLUGINS_DEFS} PARENT_SCOPE)
				set(EVOENGINE_PLUGINS_RUNTIME_LIBRARIES ${EVOENGINE_PLUGINS_RUNTIME_LIBRARIES} PARENT_SCOPE)
				set(EVOENGINE_PLUGINS_PDBS ${EVOENGINE_PLUGINS_PDBS} PARENT_SCOPE)
				set(EVOENGINE_PLUGINS_RESOURCES ${EVOENGINE_PLUGINS_RESOURCES} PARENT_SCOPE)
				return()
			endif()
		endforeach()

		if (EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/${plugin_name})
			add_subdirectory(${plugin_name})
			if(TARGET ${plugin_name}Plugin)
				if (CMAKE_CXX_COMPILER_ID STREQUAL GNU)
					message(STATUS "Setting G++ flags")
					target_compile_options(${plugin_name}Plugin PRIVATE -Werror)
				elseif (MSVC)
					message(STATUS "Setting MSVC flags")
					target_compile_options(${plugin_name}Plugin PRIVATE /EHsc /W2 /FS /c)
				elseif (CMAKE_CXX_COMPILER_ID STREQUAL Clang)
					message(STATUS "Setting Clang flags")
					target_compile_options(${plugin_name}Plugin PRIVATE -fPIC --no-warnings -Wno-c++98-compat -Wno-c++98-compat-pedantic -Wno-newline-eof -Wno-padded -Wno-exit-time-destructors -Wno-global-constructors -Wno-constant-conversion)
				endif ()
				message("Enabled [${plugin_name}] Plugin.")
				set_target_properties(${plugin_name}Plugin PROPERTIES FOLDER "Plugins")
				set(EvoEngine_Plugins ${EvoEngine_Plugins} ${plugin_name}Plugin)
			else()
				message("[${plugin_name}] Plugin not loaded.")
			endif()
		else()
			message("[${plugin_name}] Plugin directory not found.")
		endif()
	else()
		message("Disabled [${plugin_name}] Plugin.")
	endif()

	set_property(GLOBAL APPEND PROPERTY EVOENGINE_REGISTERED_PLUGIN_NAMES ${plugin_name})
	set(EvoEngine_Plugins ${EvoEngine_Plugins} PARENT_SCOPE)
	set(EVOENGINE_PLUGINS_INCLUDES ${EVOENGINE_PLUGINS_INCLUDES} PARENT_SCOPE)
	set(EVOENGINE_PLUGINS_PCHS ${EVOENGINE_PLUGINS_PCHS} PARENT_SCOPE)
	set(EVOENGINE_PLUGINS_DEFS ${EVOENGINE_PLUGINS_DEFS} PARENT_SCOPE)
	set(EVOENGINE_PLUGINS_RUNTIME_LIBRARIES ${EVOENGINE_PLUGINS_RUNTIME_LIBRARIES} PARENT_SCOPE)
	set(EVOENGINE_PLUGINS_PDBS ${EVOENGINE_PLUGINS_PDBS} PARENT_SCOPE)
	set(EVOENGINE_PLUGINS_RESOURCES ${EVOENGINE_PLUGINS_RESOURCES} PARENT_SCOPE)
endfunction()
