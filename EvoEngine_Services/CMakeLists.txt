function(evoengine_discover_services out_services)
	file(GLOB service_cmake_files
		LIST_DIRECTORIES false
		CONFIGURE_DEPENDS
		"${CMAKE_CURRENT_SOURCE_DIR}/*/CMakeLists.txt")

	set(discovered_services "")
	foreach(service_cmake_file ${service_cmake_files})
		get_filename_component(service_dir "${service_cmake_file}" DIRECTORY)
		get_filename_component(service_name "${service_dir}" NAME)
		if (NOT service_name MATCHES "^\\.")
			list(APPEND discovered_services ${service_name})
		endif()
	endforeach()
	list(SORT discovered_services)
	set(${out_services} ${discovered_services} PARENT_SCOPE)
endfunction()

function(evoengine_load_service_metadata service_name out_default_enable out_supported_platforms out_dependencies)
	set(default_enable ON)
	set(supported_platforms ALL)
	set(dependencies "")
	set(service_metadata_file "${CMAKE_CURRENT_SOURCE_DIR}/${service_name}/ServiceInfo.cmake")

	if (EXISTS "${service_metadata_file}")
		set(EVOENGINE_SERVICE_DEFAULT_ENABLE ${default_enable})
		set(EVOENGINE_SERVICE_SUPPORTED_PLATFORMS ${supported_platforms})
		set(EVOENGINE_SERVICE_DEPENDS ${dependencies})
		include("${service_metadata_file}")
		set(default_enable ${EVOENGINE_SERVICE_DEFAULT_ENABLE})
		set(supported_platforms ${EVOENGINE_SERVICE_SUPPORTED_PLATFORMS})
		set(dependencies ${EVOENGINE_SERVICE_DEPENDS})
	endif()

	set(${out_default_enable} ${default_enable} PARENT_SCOPE)
	set(${out_supported_platforms} ${supported_platforms} PARENT_SCOPE)
	set(${out_dependencies} ${dependencies} PARENT_SCOPE)
endfunction()

function(evoengine_service_platform_supported supported_platforms out_supported)
	if (NOT supported_platforms OR supported_platforms STREQUAL "ALL")
		set(${out_supported} ON PARENT_SCOPE)
		return()
	endif()

	set(is_supported OFF)
	foreach(platform_name ${supported_platforms})
		string(TOUPPER "${platform_name}" platform_name_upper)
		if ((platform_name_upper STREQUAL "WINDOWS" OR platform_name_upper STREQUAL "WIN32") AND WIN32)
			set(is_supported ON)
		elseif ((platform_name_upper STREQUAL "LINUX" OR platform_name_upper STREQUAL "UNIX") AND UNIX AND NOT APPLE)
			set(is_supported ON)
		elseif ((platform_name_upper STREQUAL "MACOS" OR platform_name_upper STREQUAL "APPLE") AND APPLE)
			set(is_supported ON)
		endif()
	endforeach()

	set(${out_supported} ${is_supported} PARENT_SCOPE)
endfunction()

function(register_evoengine_service service_name)
	get_property(registered_services GLOBAL PROPERTY EVOENGINE_REGISTERED_SERVICE_NAMES)
	if (service_name IN_LIST registered_services)
		return()
	endif()

	get_property(registering_services GLOBAL PROPERTY EVOENGINE_REGISTERING_SERVICE_NAMES)
	if (service_name IN_LIST registering_services)
		message(FATAL_ERROR "Service dependency cycle detected while registering [${service_name}].")
	endif()
	set_property(GLOBAL APPEND PROPERTY EVOENGINE_REGISTERING_SERVICE_NAMES ${service_name})

	evoengine_load_service_metadata(${service_name} default_enable supported_platforms service_dependencies)
	set(option_name EVOENGINE_ENABLE_${service_name}_SERVICE)
	option(${option_name} "Enable ${service_name} Service." ${default_enable})

	evoengine_service_platform_supported("${supported_platforms}" service_platform_supported)
	if (NOT service_platform_supported)
		message("Skipping [${service_name}] Service: supported platforms are [${supported_platforms}].")
		set_property(GLOBAL APPEND PROPERTY EVOENGINE_REGISTERED_SERVICE_NAMES ${service_name})
		set(EvoEngine_Services ${EvoEngine_Services} PARENT_SCOPE)
		set(EVOENGINE_SERVICES_INCLUDES ${EVOENGINE_SERVICES_INCLUDES} PARENT_SCOPE)
		set(EVOENGINE_SERVICES_PCHS ${EVOENGINE_SERVICES_PCHS} PARENT_SCOPE)
		set(EVOENGINE_SERVICES_DEFS ${EVOENGINE_SERVICES_DEFS} PARENT_SCOPE)
		set(EVOENGINE_SERVICES_RUNTIME_LIBRARIES ${EVOENGINE_SERVICES_RUNTIME_LIBRARIES} PARENT_SCOPE)
		set(EVOENGINE_SERVICES_PDBS ${EVOENGINE_SERVICES_PDBS} PARENT_SCOPE)
		set(EVOENGINE_SERVICES_RESOURCES ${EVOENGINE_SERVICES_RESOURCES} PARENT_SCOPE)
		return()
	endif()

	if (${option_name})
		foreach(dependency_name ${service_dependencies})
			if (dependency_name IN_LIST EVOENGINE_DISCOVERED_SERVICES)
				register_evoengine_service(${dependency_name})
			endif()
			if (NOT TARGET ${dependency_name}Service)
				message("[${service_name}] Service not loaded: dependency [${dependency_name}] Service is not available.")
				set_property(GLOBAL APPEND PROPERTY EVOENGINE_REGISTERED_SERVICE_NAMES ${service_name})
				set(EvoEngine_Services ${EvoEngine_Services} PARENT_SCOPE)
				set(EVOENGINE_SERVICES_INCLUDES ${EVOENGINE_SERVICES_INCLUDES} PARENT_SCOPE)
				set(EVOENGINE_SERVICES_PCHS ${EVOENGINE_SERVICES_PCHS} PARENT_SCOPE)
				set(EVOENGINE_SERVICES_DEFS ${EVOENGINE_SERVICES_DEFS} PARENT_SCOPE)
				set(EVOENGINE_SERVICES_RUNTIME_LIBRARIES ${EVOENGINE_SERVICES_RUNTIME_LIBRARIES} PARENT_SCOPE)
				set(EVOENGINE_SERVICES_PDBS ${EVOENGINE_SERVICES_PDBS} PARENT_SCOPE)
				set(EVOENGINE_SERVICES_RESOURCES ${EVOENGINE_SERVICES_RESOURCES} PARENT_SCOPE)
				return()
			endif()
		endforeach()

		if (EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/${service_name})
			add_subdirectory(${service_name})
			if(TARGET ${service_name}Service)
				if (CMAKE_CXX_COMPILER_ID STREQUAL GNU)
					message(STATUS "Setting G++ flags")
					target_compile_options(${service_name}Service PRIVATE -Werror)
				elseif (MSVC)
					message(STATUS "Setting MSVC flags")
					target_compile_options(${service_name}Service PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/EHsc /W2 /FS>)
				elseif (CMAKE_CXX_COMPILER_ID STREQUAL Clang)
					message(STATUS "Setting Clang flags")
					target_compile_options(${service_name}Service PRIVATE -fPIC --no-warnings -Wno-c++98-compat -Wno-c++98-compat-pedantic -Wno-newline-eof -Wno-padded -Wno-exit-time-destructors -Wno-global-constructors -Wno-constant-conversion)
				endif ()
				message("Enabled [${service_name}] Service.")
				set_target_properties(${service_name}Service PROPERTIES FOLDER "Services")
				set(EvoEngine_Services ${EvoEngine_Services} ${service_name}Service)
			else()
				message("[${service_name}] Service not loaded.")
			endif()
		else()
			message("[${service_name}] Service directory not found.")
		endif()
	else()
		message("Disabled [${service_name}] Service.")
	endif()

	set_property(GLOBAL APPEND PROPERTY EVOENGINE_REGISTERED_SERVICE_NAMES ${service_name})
	set(EvoEngine_Services ${EvoEngine_Services} PARENT_SCOPE)
	set(EVOENGINE_SERVICES_INCLUDES ${EVOENGINE_SERVICES_INCLUDES} PARENT_SCOPE)
	set(EVOENGINE_SERVICES_PCHS ${EVOENGINE_SERVICES_PCHS} PARENT_SCOPE)
	set(EVOENGINE_SERVICES_DEFS ${EVOENGINE_SERVICES_DEFS} PARENT_SCOPE)
	set(EVOENGINE_SERVICES_RUNTIME_LIBRARIES ${EVOENGINE_SERVICES_RUNTIME_LIBRARIES} PARENT_SCOPE)
	set(EVOENGINE_SERVICES_PDBS ${EVOENGINE_SERVICES_PDBS} PARENT_SCOPE)
	set(EVOENGINE_SERVICES_RESOURCES ${EVOENGINE_SERVICES_RESOURCES} PARENT_SCOPE)
endfunction()
